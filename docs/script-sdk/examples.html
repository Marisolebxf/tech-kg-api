def paper_author_person(table, row, batch) -> list[EntityRecord]:
    aid = row.get("author_id")
    paper_id = row.get("paper_id")
    # 旧口径：paper_id 去掉 __数字 后缀后与 author_id 均非空才建点
    if not aid or not _SUFFIX_RE.sub("", str(paper_id or "")):
        return []
    vid = f"person_{aid}"
    return [EntityRecord("Person", vid, {
        "name_zh": paper_text(row.get("zh_name")),
        "name_en": paper_text(row.get("en_name")),
        "extra_json": extra_json(row),
        **provenance(table=table, record_id=vid, ingest_batch=batch),
    })]

# ② build_sources：把「参数 → sources + dedupe」的决策收成一份

例二：authored_by_relation —— rank 边脚本

backend/script/relation_extractors_one_relation/authored_by_relation.py（99 行， 去掉文档字符串和双入口样板后核心只有 ~30 行）。整个脚本展示 rank 边的最小形态：

TABLES = ("dwd_zh_author", "dwd_en_author")

def authored_by(table: str, row: dict, batch: str) -> list[EdgeRecord]:
    pid = paper_source_id(row.get("paper_id"))   # resolvers：去 __数字 后缀
    aid = str(row.get("author_id")) if row.get("author_id") else ""
    if not pid or not aid:
        return []
    return [EdgeRecord(
        "AUTHORED_BY",
        f"paper_{pid}",      # 端点公式 = 实体侧公式（直接 import）
        f"person_{aid}",
        {"author_order": ..., "is_corresponding": ..., "confidence": 1.0},
        rank=0,               # 同端点唯一边：重跑覆盖更新
    )]

def build_sources(payload):
    table_choice = payload.get("table", "all")
    tables = TABLES if table_choice == "all" else (table_choice,)
    return [(t, f"SELECT * FROM {t} ORDER BY paper_id, author_id",
                 authored_by) for t in tables]

注意它不做的事：不建 Person/Paper 点（person_entity --source paper-author 和 paper_entity 先行）、不 ensure_edge_schema（AUTHORED_BY 的 schema 由 init_paper_journal_schema.py 建好）、端点不验存（旧口径信任上游已写点）。

# mapper 工厂：把 matcher/report 闭包进 mapper
def make_funded_by_mapper(matcher, report):
    def funded_by(table, row, batch):
        project_id = str(row.get("id") or "")
        if not project_id:
            return []
        institution = normalize_text(row.get("funded_institution")).rstrip("；;")
        records = []
        if institution:
            report.increment("organization_candidates")
            org_result = matcher.organization.match(institution, method="name_exact")

            # matched → 建边；ambiguous/not_found → 复核目录
            target = _matched_vid(report, org_result, "organization",
                                  {"project_id": project_id,
                                   "field": "funded_institution",
                                   "value": institution})
            if target:
                report.increment("edges_FUNDED_BY")
                records.append(EdgeRecord(
                    "FUNDED_BY", f"project_{project_id}", target,
                    {**edge_provenance(source_table=table,
                                       source_record_id=project_id,  # merge 幂等键
                                       ingest_batch=batch),
                     "funded_amount": to_float(row.get("funded_amount")),
                     **match_audit_props(org_result.method, org_result.evidence),
                     **funded_by_org_props(matcher.organization_id(target))},
                    source_tag="Project", target_tag="Organization",
                ))

        # 跨域候选只记报告不建边（PARTICIPATES_IN 归机构域脚本）
        for name in sorted(set(parse_list(row.get("participating_institution")))):
            report.add("cross_domain", {...})
        return records
    return funded_by

新脚本自查清单

写一个新抽取脚本，过一遍这张清单：

  1. 口径：实体属于哪个域 → 文本/数值/extra_json 用哪套语义（对照表）。
  2. VID：有主键 → 前缀拼接；无主键 → 复合身份哈希 / 行稳定键；可能超长 → bounded_vid。
  3. 溯源：provenance / org_provenance / edge_provenance 选一个；source_record_id 必须稳定。
  4. schema：目标 Tag / Edge 已存在？属性名与 schema 完全一致？边属性要补列就 ensure_edge_schema。
  5. 写通道：边选 rank 还是 merge（对照表）；机构域实体置 merge_protect=True。
  6. 端点：边端点公式 import 自实体侧；是否验存；悬空桩端点 validate_endpoints=False。
  7. 过滤：缺主键行返回 []；机构域记得 is_virtual_source_row。
  8. 排序：源 SQL 带确定性 ORDER BY；大表考虑 cursor_column。
  9. 双入口：build_sources(payload) 一份；main() 用 vars(args)；payload key 全 snake_case。
  10. 验证：--dry-run --limit 10 先跑一遍看 summary；再 --limit 1000 小批量真写，图上抽查点/边属性。
